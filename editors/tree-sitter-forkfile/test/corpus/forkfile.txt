  workdir app

    port 8080
    internal_ports [8081, 8082]
    route /api strip-prefix domain main
    run: @pnpm dev --port $port
    pre-deploy: echo $env.DATABASE_URL
  }
}

  image tools from alpine:3.20 {
    service shell {
      port 9000
      route exact /healthz
    }
  }
}
