  staging

  version 18
  database analytics
}

redis cache

s3 assets {
  policy public-read
  cdn static-cdn
}

env shared {
  DATABASE_URL = "postgres://localhost"
  TOKEN = secret(DB_TOKEN) ?? random_hex(32)
}

image backend . {
  dockerfile Dockerfile
  workdir app
  arg BUILD_MODE = dev ?? "release"
  arg VERSION = "1.0"
  use env shared
  before-start when changed(package.json): @pnpm install
  service web {
    port 8080
    internal_ports [8081, 8082]
    route /api strip-prefix domain main
    run: @pnpm dev --port $port
    pre-deploy: echo $env.DATABASE_URL
  }
}

on !dev {
  image tools from alpine:3.20 {
    service shell {
      port 9000
      route exact /healthz
    }
  }
}
